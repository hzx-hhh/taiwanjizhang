import json import os import uuid import tkinter as tk from tkinter import ttk, messagebox, filedialog import datetime # 账本数据文件路径 DATA_FILE = "account_books.json" # 当前激活的账本ID ACTIVE_BOOK_ID = None # 账本数据结构 account_books = { "active_book_id": None, "books": {} } # 初始化账本数据 def init_data(): global account_books, ACTIVE_BOOK_ID if os.path.exists(DATA_FILE): try: with open(DATA_FILE, 'r', encoding='utf-8') as f: account_books = json.load(f) ACTIVE_BOOK_ID = account_books.get("active_book_id") # 如果没有激活的账本，创建一个默认账本 if not ACTIVE_BOOK_ID or ACTIVE_BOOK_ID not in account_books["books"]: create_new_book("默认账本") except Exception as e: messagebox.showerror("错误", f"加载账本数据失败：{str(e)}") create_new_book("默认账本") else: create_new_book("默认账本") # 保存账本数据 def save_data(): try: with open(DATA_FILE, 'w', encoding='utf-8') as f: json.dump(account_books, f, ensure_ascii=False, indent=2) except Exception as e: messagebox.showerror("错误", f"保存账本数据失败：{str(e)}") # 创建新账本 def create_new_book(book_name): global ACTIVE_BOOK_ID book_id = str(uuid.uuid4()) account_books["books"][book_id] = { "name": book_name, "id": book_id, "records": [], "create_time": datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S") } account_books["active_book_id"] = book_id ACTIVE_BOOK_ID = book_id save_data() update_book_list() update_record_list() update_book_id_display() # 更新账本列表显示 def update_book_list(): book_list.delete(0, tk.END) for book_id, book_info in account_books["books"].items(): if book_id == ACTIVE_BOOK_ID: book_list.insert(tk.END, f"★ {book_info['name']}") else: book_list.insert(tk.END, book_info['name']) # 更新账本ID显示 def update_book_id_display(): if ACTIVE_BOOK_ID and ACTIVE_BOOK_ID in account_books["books"]: book_id_label.config(text=f"当前账本ID：{ACTIVE_BOOK_ID}") else: book_id_label.config(text="当前账本ID：无") # 更新记录列表显示 def update_record_list(): record_list.delete(0, tk.END) if ACTIVE_BOOK_ID and ACTIVE_BOOK_ID in account_books["books"]: records = account_books["books"][ACTIVE_BOOK_ID]["records"] # 按时间倒序显示 for record in sorted(records, key=lambda x: x["time"], reverse=True): record_type = "收入" if record["type"] == "income" else "支出" record_list.insert(tk.END, f"{record['time']} | {record_type} | {record['category']} | {record['amount']}元 | {record['remark']}") # 添加收支记录 def add_record(): if not ACTIVE_BOOK_ID: messagebox.showwarning("警告", "请先选择一个账本！") return try: amount = float(amount_entry.get()) if amount <= 0: messagebox.showwarning("警告", "金额必须大于0！") return record_type = type_var.get() category = category_entry.get().strip() remark = remark_entry.get().strip() if not category: messagebox.showwarning("警告", "请输入分类！") return record = { "id": str(uuid.uuid4()), "type": record_type, "category": category, "amount": amount, "remark": remark, "time": datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S") } account_books["books"][ACTIVE_BOOK_ID]["records"].append(record) save_data() update_record_list() # 清空输入框 amount_entry.delete(0, tk.END) category_entry.delete(0, tk.END) remark_entry.delete(0, tk.END) messagebox.showinfo("成功", "记录添加成功！") except ValueError: messagebox.showerror("错误", "请输入有效的金额！") # 切换账本 def switch_book(event=None): global ACTIVE_BOOK_ID selected_index = book_list.curselection() if selected_index: # 找到选中的账本ID book_names = list(account_books["books"].keys()) for i, (book_id, book_info) in enumerate(account_books["books"].items()): if i == selected_index[0]: ACTIVE_BOOK_ID = book_id account_books["active_book_id"] = book_id save_data() update_book_list() update_record_list() update_book_id_display() break # 删除选中的记录 def delete_record(): selected_index = record_list.curselection() if not selected_index: messagebox.showwarning("警告", "请选择要删除的记录！") return if messagebox.askyesno("确认", "确定要删除选中的记录吗？"): records = account_books["books"][ACTIVE_BOOK_ID]["records"] # 按时间倒序后的索引需要转换 sorted_records = sorted(records, key=lambda x: x["time"], reverse=True) record_id = sorted_records[selected_index[0]]["id"] # 找到原始记录并删除 for i, record in enumerate(records): if record["id"] == record_id: del records[i] break save_data() update_record_list() messagebox.showinfo("成功", "记录删除成功！") # 账本管理窗口 def manage_books(): manage_window = tk.Toplevel(root) manage_window.title("账本管理") manage_window.geometry("400x300") manage_window.resizable(False, False) # 账本列表 list_frame = ttk.Frame(manage_window) list_frame.pack(pady=10, padx=10, fill=tk.BOTH, expand=True) manage_book_list = tk.Listbox(list_frame, width=50, height=10) manage_book_list.pack(side=tk.LEFT, fill=tk.BOTH, expand=True) scrollbar = ttk.Scrollbar(list_frame, orient=tk.VERTICAL, command=manage_book_list.yview) scrollbar.pack(side=tk.RIGHT, fill=tk.Y) manage_book_list.config(yscrollcommand=scrollbar.set) # 填充账本列表 for book_id, book_info in account_books["books"].items(): if book_id == ACTIVE_BOOK_ID: manage_book_list.insert(tk.END, f"★ {book_info['name']}") else: manage_book_list.insert(tk.END, book_info['name']) # 按钮区域 btn_frame = ttk.Frame(manage_window) btn_frame.pack(pady=10) # 重命名账本 def rename_book(): selected_index = manage_book_list.curselection() if not selected_index: messagebox.showwarning("警告", "请选择要重命名的账本！") return # 找到选中的账本ID for i, (book_id, book_info) in enumerate(account_books["books"].items()): if i == selected_index[0]: new_name = tk.simpledialog.askstring("重命名", "请输入新的账本名称：", initialvalue=book_info["name"]) if new_name and new_name.strip(): account_books["books"][book_id]["name"] = new_name.strip() save_data() update_book_list() # 更新管理窗口的列表 manage_book_list.delete(0, tk.END) for bid, binfo in account_books["books"].items(): if bid == ACTIVE_BOOK_ID: manage_book_list.insert(tk.END, f"★ {binfo['name']}") else: manage_book_list.insert(tk.END, binfo['name']) break # 删除账本 def delete_book(): selected_index = manage_book_list.curselection() if not selected_index: messagebox.showwarning("警告", "请选择要删除的账本！") return # 找到选中的账本ID book_ids = list(account_books["books"].keys()) selected_book_id = book_ids[selected_index[0]] if selected_book_id == ACTIVE_BOOK_ID: messagebox.showwarning("警告", "不能删除当前正在使用的账本！") return if len(account_books["books"]) <= 1: messagebox.showwarning("警告", "至少保留一个账本！") return if messagebox.askyesno("确认", "确定要删除选中的账本吗？此操作不可恢复！"): del account_books["books"][selected_book_id] save_data() update_book_list() # 更新管理窗口的列表 manage_book_list.delete(0, tk.END) for bid, binfo in account_books["books"].items(): if bid == ACTIVE_BOOK_ID: manage_book_list.insert(tk.END, f"★ {binfo['name']}") else: manage_book_list.insert(tk.END, binfo['name']) # 新建账本 def new_book(): book_name = tk.simpledialog.askstring("新建账本", "请输入账本名称：") if book_name and book_name.strip(): create_new_book(book_name.strip()) # 更新管理窗口的列表 manage_book_list.delete(0, tk.END) for bid, binfo in account_books["books"].items(): if bid == ACTIVE_BOOK_ID: manage_book_list.insert(tk.END, f"★ {binfo['name']}") else: manage_book_list.insert(tk.END, binfo['name']) ttk.Button(btn_frame, text="新建账本", command=new_book).grid(row=0, column=0, padx=5) ttk.Button(btn_frame, text="重命名", command=rename_book).grid(row=0, column=1, padx=5) ttk.Button(btn_frame, text="删除账本", command=delete_book).grid(row=0, column=2, padx=5) # 导出账本为JSON def export_book(): if not ACTIVE_BOOK_ID: messagebox.showwarning("警告", "请先选择一个账本！") return file_path = filedialog.asksaveasfilename( defaultextension=".json", filetypes=[("JSON文件", "*.json"), ("所有文件", "*.*")], title="导出账本" ) if file_path: try: book_data = account_books["books"][ACTIVE_BOOK_ID] with open(file_path, 'w', encoding='utf-8') as f: json.dump(book_data, f, ensure_ascii=False, indent=2) messagebox.showinfo("成功", "账本导出成功！") except Exception as e: messagebox.showerror("错误", f"导出失败：{str(e)}") # 导入JSON账本 def import_book(): file_path = filedialog.askopenfilename( filetypes=[("JSON文件", "*.json"), ("所有文件", "*.*")], title="导入账本" ) if file_path: try: with open(file_path, 'r', encoding='utf-8') as f: book_data = json.load(f) # 验证账本数据格式 if not isinstance(book_data, dict) or "name" not in book_data or "records" not in book_data: messagebox.showerror("错误", "导入的文件不是有效的账本数据！") return # 生成新的账本ID new_book_id = str(uuid.uuid4()) book_data["id"] = new_book_id if "create_time" not in book_data: book_data["create_time"] = datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S") account_books["books"][new_book_id] = book_data save_data() update_book_list() messagebox.showinfo("成功", f"账本「{book_data['name']}」导入成功！") except Exception as e: messagebox.showerror("错误", f"导入失败：{str(e)}") # 共享账本（简化版，直接复制ID到剪贴板） def share_book(): if not ACTIVE_BOOK_ID: messagebox.showwarning("警告", "请先选择一个账本！") return # 直接复制账本ID到剪贴板 root.clipboard_clear() root.clipboard_append(ACTIVE_BOOK_ID) root.update() # 保持剪贴板内容 messagebox.showinfo("成功", f"账本ID已复制到剪贴板：\n{ACTIVE_BOOK_ID}") # 粘贴共享ID并同步账本 def paste_share_id(): share_id = tk.simpledialog.askstring("同步账本", "请粘贴共享ID：") if not share_id: return share_id = share_id.strip() if share_id == ACTIVE_BOOK_ID: messagebox.showinfo("提示", "这是当前正在使用的账本ID！") return if share_id in account_books["books"]: # 切换到该账本 global ACTIVE_BOOK_ID ACTIVE_BOOK_ID = share_id account_books["active_book_id"] = share_id save_data() update_book_list() update_record_list() update_book_id_display() messagebox.showinfo("成功", f"已切换到共享账本：{account_books['books'][share_id]['name']}") else: messagebox.showwarning("警告", "未找到该ID对应的账本！") # 同步账本（简化版，仅检查是否有更新） def sync_book(): if not ACTIVE_BOOK_ID: messagebox.showwarning("警告", "请先选择一个账本！") return # 实际应用中这里应该是网络同步逻辑 # 这里简化为提示同步完成 messagebox.showinfo("同步完成", "账本同步检查完成！\n当前账本ID：" + ACTIVE_BOOK_ID) # 创建主窗口 root = tk.Tk() root.title("简易账本管理系统") root.geometry("800x600") root.resizable(True, True) # 导入tkinter.simpledialog（用于输入框） import tkinter.simpledialog as simpledialog # 主框架 main_frame = ttk.PanedWindow(root, orient=tk.HORIZONTAL) main_frame.pack(fill=tk.BOTH, expand=True, padx=10, pady=10) # 左侧账本列表区域 left_frame = ttk.Frame(main_frame, width=200) main_frame.add(left_frame, weight=1) # 账本列表标题 ttk.Label(left_frame, text="账本列表", font=("Arial", 12, "bold")).pack(pady=5) # 账本列表 book_list_frame = ttk.Frame(left_frame) book_list_frame.pack(fill=tk.BOTH, expand=True, padx=5, pady=5) book_list = tk.Listbox(book_list_frame) book_list.pack(side=tk.LEFT, fill=tk.BOTH, expand=True) book_list_scroll = ttk.Scrollbar(book_list_frame, orient=tk.VERTICAL, command=book_list.yview) book_list_scroll.pack(side=tk.RIGHT, fill=tk.Y) book_list.config(yscrollcommand=book_list_scroll.set) book_list.bind('', switch_book) # 账本ID显示 book_id_label = ttk.Label(left_frame, text="当前账本ID：无", font=("Arial", 9), foreground="gray") book_id_label.pack(pady=5, padx=5, anchor=tk.W) # 账本管理按钮 ttk.Button(left_frame, text="账本管理", command=manage_books).pack(pady=5, fill=tk.X, padx=5) #
